// background/service-worker.js
import { initStorage } from './storage-manager.js';
import { initStatisticsManager, updateTotalBlocks, getStatistics } from './statistics-manager.js';
import { updateRedirectRules } from './redirect-rules.js';
import { createContextMenu, handleContextMenuClick } from './context-menu.js';
import { initAlarms, onAlarm } from './alarms.js';
import { MESSAGE_TYPES, STORAGE_KEYS } from '../shared/constants.js';

async function initBackground() {
  await initStorage();
  await initStatisticsManager();
  await updateRedirectRules();
  createContextMenu();
  await initAlarms();
  console.log('[Background] YouTube Blocker initialised');
}

chrome.storage.onChanged.addListener(async (changes) => {
  if (changes.blockAllShorts || changes.blockedChannels || changes.forceDesktopMode) {
    await updateRedirectRules();
  }
});

chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  if (message.type === MESSAGE_TYPES.BLOCKS_COUNT) {
    updateTotalBlocks(message.delta);
    sendResponse({ success: true });
    return false;
  }
  if (message.type === MESSAGE_TYPES.GET_STATS) {
    getStatistics().then(sendResponse);
    return true;
  }
  if (message.action === 'adBlocked') {
    updateAdStats();
    sendResponse({ success: true });
    return false;
  }
  return false;
});

chrome.contextMenus.onClicked.addListener(handleContextMenuClick);
chrome.alarms.onAlarm.addListener(onAlarm);

// ========== Batched Ad Statistics ==========
let pendingAdIncrement = 0;
let adStatsTimer = null;

async function flushAdStats() {
  if (pendingAdIncrement === 0) return;
  const delta = pendingAdIncrement;
  pendingAdIncrement = 0;
  const today = new Date().toDateString();
  const { adsBlockedToday = 0, adsBlockedTotal = 0, adsLastDate = today } =
    await chrome.storage.sync.get([STORAGE_KEYS.adsBlockedToday, STORAGE_KEYS.adsBlockedTotal, STORAGE_KEYS.adsLastDate]);

  if (today !== adsLastDate) {
    await chrome.storage.sync.set({
      adsBlockedToday: delta,
      adsBlockedTotal: adsBlockedTotal + delta,
      adsLastDate: today
    });
  } else {
    await chrome.storage.sync.set({
      adsBlockedToday: adsBlockedToday + delta,
      adsBlockedTotal: adsBlockedTotal + delta
    });
  }
}

function updateAdStats() {
  pendingAdIncrement++;
  if (adStatsTimer) clearTimeout(adStatsTimer);
  adStatsTimer = setTimeout(flushAdStats, 2000);
}

// ========== Whitelist Homepage Redirect to Search ==========
chrome.tabs.onUpdated.addListener(async (tabId, changeInfo, tab) => {
  if (changeInfo.status !== 'complete') return;
  const url = tab.url;
  if (!url) return;

  const isHomepage = /^(https?:\/\/)?(www\.|m\.)?youtube\.com\/?$/.test(url);
  if (!isHomepage) return;

  const { whitelistMode = false, whitelistSearchRedirect = false, whitelistSearchSelected = '' } =
    await chrome.storage.sync.get([
      'whitelistMode',
      STORAGE_KEYS.whitelistSearchRedirect,
      STORAGE_KEYS.whitelistSearchSelected
    ]);

  if (!whitelistMode || !whitelistSearchRedirect || !whitelistSearchSelected) return;
  if (url.includes('/results?search_query=')) return;

  const isMobile = url.includes('m.youtube.com');
  const base = isMobile ? 'https://m.youtube.com' : 'https://www.youtube.com';
  const searchUrl = `${base}/results?search_query=${encodeURIComponent(whitelistSearchSelected)}`;
  chrome.tabs.update(tabId, { url: searchUrl });
});

initBackground();

// background/storage-manager.js
import { STORAGE_KEYS, DEFAULTS, getTodayString } from '../shared/constants.js';

/**
 * Get a single storage value by key (with default fallback)
 */
export async function getStorageItem(key) {
  const result = await chrome.storage.sync.get(key);
  return result[key] ?? DEFAULTS[key];
}

/**
 * Set a single storage value
 */
export async function setStorageItem(key, value) {
  await chrome.storage.sync.set({ [key]: value });
}

/**
 * Get multiple storage values (fills missing with defaults)
 */
export async function getStorageItems(keys) {
  const result = await chrome.storage.sync.get(keys);
  for (const key of keys) {
    if (result[key] === undefined) {
      result[key] = DEFAULTS[key];
    }
  }
  return result;
}

/**
 * Set multiple storage values at once
 */
export async function setStorageItems(items) {
  await chrome.storage.sync.set(items);
}

/**
 * Initialise storage with default values (only sets missing keys)
 */
export async function initStorage() {
  const allKeys = Object.values(STORAGE_KEYS);
  const existing = await chrome.storage.sync.get(allKeys);
  const toSet = {};
  for (const key of allKeys) {
    if (existing[key] === undefined) {
      let defaultValue = DEFAULTS[key];
      if (key === STORAGE_KEYS.dailyStats && typeof defaultValue === 'object') {
        defaultValue = { date: getTodayString(), count: 0 };
      }
      toSet[key] = defaultValue;
    }
  }
  if (Object.keys(toSet).length) {
    await chrome.storage.sync.set(toSet);
  }
}

// background/statistics-manager.js
import { STORAGE_KEYS, getTodayString } from '../shared/constants.js';
import { getStorageItem, setStorageItem } from './storage-manager.js';

let totalBlocksCache = 0;
let dailyStatsCache = null;
let weeklyStatsCache = null;

function updateBadge(total) {
  let text = '';
  if (total >= 1000) text = (total / 1000).toFixed(1) + 'k';
  else if (total > 0) text = total.toString();
  chrome.action.setBadgeText({ text });
  chrome.action.setBadgeBackgroundColor({ color: '#667eea' });
}

export async function rolloverDailyStats() {
  const today = getTodayString();
  let daily = await getStorageItem(STORAGE_KEYS.dailyStats);
  let weekly = await getStorageItem(STORAGE_KEYS.weeklyStats);

  if (daily.date !== today) {
    weekly.push({ date: daily.date, count: daily.count });
    while (weekly.length > 7) weekly.shift();
    await setStorageItem(STORAGE_KEYS.weeklyStats, weekly);
    daily = { date: today, count: 0 };
    await setStorageItem(STORAGE_KEYS.dailyStats, daily);
    dailyStatsCache = daily;
    weeklyStatsCache = weekly;
  }
}

export async function updateTotalBlocks(delta) {
  if (delta <= 0) return;

  let total = totalBlocksCache;
  if (total === undefined) total = await getStorageItem(STORAGE_KEYS.totalBlocks);
  total += delta;
  totalBlocksCache = total;
  await setStorageItem(STORAGE_KEYS.totalBlocks, total);
  updateBadge(total);

  let daily = dailyStatsCache;
  if (!daily) daily = await getStorageItem(STORAGE_KEYS.dailyStats);
  const today = getTodayString();
  if (daily.date !== today) {
    await rolloverDailyStats();
    daily = { date: today, count: 0 };
  }
  daily.count += delta;
  dailyStatsCache = daily;
  await setStorageItem(STORAGE_KEYS.dailyStats, daily);
}

export async function getStatistics() {
  const total = totalBlocksCache !== undefined ? totalBlocksCache : await getStorageItem(STORAGE_KEYS.totalBlocks);
  const daily = dailyStatsCache || await getStorageItem(STORAGE_KEYS.dailyStats);
  const weekly = weeklyStatsCache || await getStorageItem(STORAGE_KEYS.weeklyStats);
  return { totalBlocks: total, dailyStats: daily, weeklyStats: weekly };
}

export async function initStatisticsManager() {
  totalBlocksCache = await getStorageItem(STORAGE_KEYS.totalBlocks);
  dailyStatsCache = await getStorageItem(STORAGE_KEYS.dailyStats);
  weeklyStatsCache = await getStorageItem(STORAGE_KEYS.weeklyStats);
  updateBadge(totalBlocksCache);
  await rolloverDailyStats();
}

// background/redirect-rules.js
import { STORAGE_KEYS } from '../shared/constants.js';

/**
 * Update dynamic redirect rules based on current settings
 * - Rule 1 (id=1): Redirect /shorts/* to YouTube homepage if blockAllShorts is true
 * - Rule 2 (id=2): Redirect each blocked channel's @handle URL to homepage
 * - Rule 3 (id=3): Redirect m.youtube.com to www.youtube.com if forceDesktopMode is true
 */
export async function updateRedirectRules() {
  const { blockAllShorts = false, blockedChannels = [], forceDesktopMode = false } =
    await chrome.storage.sync.get([
      STORAGE_KEYS.blockAllShorts,
      STORAGE_KEYS.blockedChannels,
      STORAGE_KEYS.forceDesktopMode
    ]);

  const newRules = [];

  // 1. Shorts redirect (desktop + mobile)
  if (blockAllShorts) {
    newRules.push({
      id: 1,
      priority: 1,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.REDIRECT,
        redirect: { url: 'https://www.youtube.com/' }
      },
      condition: {
        urlFilter: '*://www.youtube.com/shorts*',
        resourceTypes: [chrome.declarativeNetRequest.ResourceType.MAIN_FRAME]
      }
    });
    newRules.push({
      id: 2,
      priority: 1,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.REDIRECT,
        redirect: { url: 'https://m.youtube.com/' }
      },
      condition: {
        urlFilter: '*://m.youtube.com/shorts*',
        resourceTypes: [chrome.declarativeNetRequest.ResourceType.MAIN_FRAME]
      }
    });
  }

  // 2. Blocked channel redirect rules (for handles)
  let nextId = 1000;
  const handles = new Set();
  for (const item of blockedChannels) {
    let handle = item;
    if (handle.includes('youtube.com/@')) {
      const match = handle.match(/youtube\.com\/@([^/?]+)/);
      if (match) handle = '@' + match[1];
      else continue;
    }
    if (handle.startsWith('@')) {
      handles.add(handle.substring(1)); // store without '@'
    }
  }
  for (const handle of handles) {
    newRules.push({
      id: nextId++,
      priority: 1,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.REDIRECT,
        redirect: { url: 'https://www.youtube.com/' }
      },
      condition: {
        urlFilter: `*://www.youtube.com/@${handle}*`,
        resourceTypes: [chrome.declarativeNetRequest.ResourceType.MAIN_FRAME]
      }
    });
    newRules.push({
      id: nextId++,
      priority: 1,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.REDIRECT,
        redirect: { url: 'https://m.youtube.com/' }
      },
      condition: {
        urlFilter: `*://m.youtube.com/@${handle}*`,
        resourceTypes: [chrome.declarativeNetRequest.ResourceType.MAIN_FRAME]
      }
    });
  }

  // 3. Force Desktop Mode
  if (forceDesktopMode) {
    newRules.push({
      id: 3,
      priority: 1,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.REDIRECT,

// background/context-menu.js
import { STORAGE_KEYS, MESSAGE_TYPES } from '../shared/constants.js';
import { notifyYouTubeTabs } from '../shared/messaging.js';

const CONTEXT_MENU_ID = 'block-channel';

/**
 * Create the context menu item (called on extension install)
 * Removes any existing item first using callback style to avoid errors.
 */
export function createContextMenu() {
  chrome.contextMenus.remove(CONTEXT_MENU_ID, () => {
    const lastError = chrome.runtime.lastError;
    if (lastError) {
      console.debug('Context menu not found, creating new one');
    }
    chrome.contextMenus.create({
      id: CONTEXT_MENU_ID,
      title: '🚫 Block this channel',
      contexts: ['link', 'video', 'image'],
      targetUrlPatterns: ['*://*.youtube.com/*', '*://youtube.com/*']
    });
  });
}

/**
 * Extract channel identifier from a YouTube URL
 * @param {string} url - The URL from the context menu event
 * @returns {string|null} - Channel identifier (@handle, channel ID, or null)
 */
function extractChannelFromUrl(url) {
  if (!url) return null;
  // Handle @handle URLs
  let match = url.match(/youtube\.com\/@([^/?]+)/);
  if (match) return '@' + match[1];
  // Handle /channel/... URLs
  match = url.match(/youtube\.com\/channel\/([^/?]+)/);
  if (match) return match[1];
  // Handle /c/... URLs
  match = url.match(/youtube\.com\/c\/([^/?]+)/);
  if (match) return match[1];
  return null;
}

/**
 * Handle context menu click: add the channel to blocklist
 * @param {object} info - Context menu event info
 */
export async function handleContextMenuClick(info) {
  if (info.menuItemId !== CONTEXT_MENU_ID) return;

  const url = info.linkUrl || info.srcUrl || '';
  const channelId = extractChannelFromUrl(url);
  if (!channelId) return;

  const { blockedChannels = [] } = await chrome.storage.sync.get(STORAGE_KEYS.blockedChannels);
  if (!blockedChannels.includes(channelId)) {
    const newList = [...blockedChannels, channelId];
    await chrome.storage.sync.set({ [STORAGE_KEYS.blockedChannels]: newList });
    await notifyYouTubeTabs(MESSAGE_TYPES.SETTINGS_UPDATED);
  }
}

  async function deriveKey(passphrase, salt) {
    const enc = new TextEncoder();
    const keyMaterial = await crypto.subtle.importKey(
      'raw', enc.encode(passphrase), 'PBKDF2', false, ['deriveBits', 'deriveKey']
    );
    return crypto.subtle.deriveKey(
      { name: 'PBKDF2', salt: enc.encode(salt), iterations: 100000, hash: 'SHA-256' },
      keyMaterial,
      { name: 'AES-GCM', length: 256 },
      false,
      ['decrypt']
    );
  }

    const key = await deriveKey(passphrase, salt);
    const iv = new Uint8Array(encryptedObj.iv);

    const decrypted = await crypto.subtle.decrypt(

// Helper: check if a video is too old based on upload date
function isVideoTooOld(videoElement) {
  if (!blockByUploadDate) return false;
  const dateElem = videoElement.querySelector('#metadata-line span:nth-child(2), .ytd-video-meta-block #metadata-line span:last-child');
  if (!dateElem) return false;
  const dateText = dateElem.textContent.trim();
  const match = dateText.match(/(\d+)\s+(day|week|month|year)s?\s+ago/);
  if (match) {
    const num = parseInt(match[1], 10);
    const unit = match[2];
    let days = 0;
    if (unit === 'day') days = num;
    else if (unit === 'week') days = num * 7;
    else if (unit === 'month') days = num * 30;
    else if (unit === 'year') days = num * 365;
    return days > maxVideoAgeDays;
  }
  return false;
}

// content/channel-detector.js
// Extracts channel display name and handle from YouTube elements

import { isMobileSite } from './dom-selectors.js';

export function getChannelInfo(element) {
  let displayName = null, handle = null;
  const isMobile = isMobileSite();
  if (isMobile) {
    const nameEl = element.querySelector('.channel-name, yt-formatted-string#channel-name');
    if (nameEl) displayName = nameEl.textContent.trim();
    const handleLink = element.querySelector('a[href^="/@"]');
    if (handleLink) {
      const match = handleLink.getAttribute('href').match(/\/@([^/?]+)/);
      if (match) handle = '@' + match[1];
    }
  } else {
    const nameSel = '#channel-name a, #owner-name a, a.ytd-channel-name, #byline-container a, #channel-title #text';
    const nameEl = element.querySelector(nameSel);
    if (nameEl) displayName = nameEl.textContent.trim();
    const handleLink = element.querySelector('a[href^="/@"]');
    if (handleLink) {
      const match = handleLink.getAttribute('href').match(/\/@([^/?]+)/);
      if (match) handle = '@' + match[1];
      else if (handleLink.textContent.trim().startsWith('@')) handle = handleLink.textContent.trim();
    }
    if (!handle && element.textContent.includes('@')) {
      const atMatch = element.textContent.match(/@[^\s]+/);
      if (atMatch) handle = atMatch[0];
    }
  }
  return { displayName, handle };
}

// content/keyword-matcher.js
// Checks video titles against keywords and whitelist keywords

let blockKeywords = [];
let enableKeywordBlocking = false;
let whitelistKeywords = new Set();
let whitelistMode = false;

export function updateKeywordSettings(keywords, keywordBlockingEnabled, whitelistKeywordsSet, whitelistModeEnabled) {
  blockKeywords = keywords || [];
  enableKeywordBlocking = keywordBlockingEnabled === true;
  whitelistKeywords = whitelistKeywordsSet || new Set();
  whitelistMode = whitelistModeEnabled === true;
}

function isWhitelistedKeyword(title) {
  if (!whitelistMode || !title) return false;
  const lowerTitle = title.toLowerCase();
  for (const kw of whitelistKeywords) if (lowerTitle.includes(kw)) return true;
  return false;
}

export function shouldBlockByKeyword(title) {
  if (!title) return false;
  const lowerTitle = title.toLowerCase();
  if (whitelistMode) {
    if (whitelistKeywords.size > 0) {
      return !isWhitelistedKeyword(lowerTitle);
    } else {
      return true; // no whitelist keywords -> block all
    }
  } else {
    if (!enableKeywordBlocking || blockKeywords.length === 0) return false;
    for (const kw of blockKeywords) if (lowerTitle.includes(kw)) return true;
    return false;
  }
}

// content/statistics-sender.js
// Sends block counts to background (batched, using session storage)

let pendingDelta = 0;
let deltaTimer = null;

export function hideElement(el) {
  if (!el || el.getAttribute('data-blocked') === 'true') return;
  el.style.setProperty('display', 'none', 'important');
  el.classList.add('yt-blocked');
  el.setAttribute('data-blocked', 'true');
  pendingDelta++;
  if (deltaTimer) clearTimeout(deltaTimer);
  deltaTimer = setTimeout(() => {
    if (pendingDelta > 0) {
      chrome.runtime.sendMessage({ type: 'BLOCKS_COUNT', delta: pendingDelta });
      pendingDelta = 0;
    }
  }, 500);
}

export function flushPendingDelta() {
  if (deltaTimer) clearTimeout(deltaTimer);
  if (pendingDelta > 0) {
    chrome.runtime.sendMessage({ type: 'BLOCKS_COUNT', delta: pendingDelta });
    pendingDelta = 0;
  }
}

// content/mutation-observer.js
// Handles MutationObserver for dynamic content and SPA navigation detection

import { getSelectorString } from './dom-selectors.js';
import { scanAllElements, scheduleScan } from './element-processor.js';

let observer = null;
let urlObserver = null;

/**
 * Start a MutationObserver to watch for added DOM nodes and trigger a rescan
 */
export function startObserver() {
  if (observer) observer.disconnect();
  observer = new MutationObserver((mutations) => {
    for (const mutation of mutations) {
      if (mutation.addedNodes.length) {
        scheduleScan();
        break;
      }
    }
  });
  observer.observe(document.body, { childList: true, subtree: true });
}

/**
 * Monitor YouTube SPA navigation (URL changes) and reset blocking state
 */
export function monitorNavigation() {
  let lastUrl = location.href;
  urlObserver = new MutationObserver(() => {
    if (location.href !== lastUrl) {
      lastUrl = location.href;
      // Reset all processed flags and hidden styles for the new page
      const elements = document.querySelectorAll(getSelectorString());
      for (const el of elements) {
        el.removeAttribute('data-processed');
        if (el.style.display === 'none') el.style.display = '';
        el.classList.remove('yt-blocked');
        el.removeAttribute('data-blocked');
      }
      // Re-scan after a short delay to let the page render
      setTimeout(scanAllElements, 500);
    }
  });
  urlObserver.observe(document, { subtree: true, childList: true });
}

// content/spinner.js
// Creates, shows, and hides the filtering spinner (dynamic text)

let spinner = null;
let hideTimeout = null;
let currentAction = 'Filtering';

/**
 * Update the spinner text (e.g., "Blocking ads...")
 * @param {string} action
 */
export function updateSpinnerText(action) {
  currentAction = action;
  if (spinner) {
    const textSpan = spinner.querySelector('.spinner-text');
    if (textSpan) textSpan.textContent = action + '...';
  }
}

/**
 * Create the spinner element and append it to the page body
 */
export function createSpinner() {
  if (document.getElementById('yt-blocker-spinner')) return;
  const div = document.createElement('div');
  div.id = 'yt-blocker-spinner';
  div.innerHTML = `<div class="spinner-icon"></div><span class="spinner-text">Filtering...</span>`;
  document.body.appendChild(div);
  spinner = div;
}

/**
 * Show the spinner (adds 'show' class) with optional action text
 * @param {string} action - e.g., 'Blocking ads'
 */
export function showSpinner(action = 'Filtering') {
  if (!spinner) createSpinner();
  updateSpinnerText(action);
  requestAnimationFrame(() => {
    if (spinner) spinner.classList.add('show');
  });
}

/**
 * Hide the spinner (removes 'show' class)
 */
export function hideSpinner() {
  if (spinner) spinner.classList.remove('show');
}

/**
 * Schedule hiding the spinner after a short delay (to avoid flickering)
 */
export function scheduleHideSpinner() {
  if (hideTimeout) clearTimeout(hideTimeout);
  hideTimeout = setTimeout(() => {
    hideSpinner();
    hideTimeout = null;
  }, 300);
}

// popup/popup.js
import { STORAGE_KEYS, MESSAGE_TYPES } from '../shared/constants.js';
import { sendToBackground, notifyYouTubeTabs } from '../shared/messaging.js';

// DOM elements
const channelInput = document.getElementById('channelInput');
const addBtn = document.getElementById('addBtn');
const blocklistUl = document.getElementById('blocklist');
const blocklistSearch = document.getElementById('blocklistSearch');
const clearAllBtn = document.getElementById('clearAllBtn');
const blocklistCount = document.getElementById('blocklistCount');
const blockAllShortsCheckbox = document.getElementById('blockAllShortsCheckbox');
const cleanSidebarCheckbox = document.getElementById('cleanSidebarCheckbox');
const backgroundPlaybackCheckbox = document.getElementById('backgroundPlaybackCheckbox');
const forceDesktopModeCheckbox = document.getElementById('forceDesktopModeCheckbox');
const totalBlocksSpan = document.getElementById('totalBlocks');
const todayBlocksSpan = document.getElementById('todayBlocks');
const openOptionsBtn = document.getElementById('openOptionsBtn');
const adBlockToggle = document.getElementById('adBlockToggle');
const videoAdsToggle = document.getElementById('videoAdsToggle');
const bannerAdsToggle = document.getElementById('bannerAdsToggle');
const adsBlockedTodaySpan = document.getElementById('adsBlockedToday');
const adsBlockedTotalSpan = document.getElementById('adsBlockedTotal');

let currentBlocklist = [];
let filteredBlocklist = [];

function extractChannelIdentifier(input) {
  let trimmed = input.trim();
  if (!trimmed) return null;
  if (trimmed.includes('youtube.com') || trimmed.includes('youtu.be')) {
    try {
      let url = new URL(trimmed);
      let match = url.pathname.match(/\/@([^/?]+)/);
      if (match) return '@' + match[1];
      match = url.pathname.match(/\/c\/([^/?]+)/);
      if (match) return match[1];
      match = url.pathname.match(/\/channel\/([^/?]+)/);
      if (match) return match[1];
    } catch(e) {}
  }
  if (trimmed.startsWith('@')) return trimmed;
  return trimmed;
}

async function loadAllSettings() {
  const result = await chrome.storage.sync.get([
    STORAGE_KEYS.blockedChannels,
    STORAGE_KEYS.blockAllShorts,
    STORAGE_KEYS.cleanSidebar,
    STORAGE_KEYS.backgroundPlayback,
    STORAGE_KEYS.forceDesktopMode,
    STORAGE_KEYS.adBlockEnabled,
    STORAGE_KEYS.blockVideoAds,
    STORAGE_KEYS.blockBannerAds
  ]);
  currentBlocklist = result[STORAGE_KEYS.blockedChannels] || [];
  filteredBlocklist = [...currentBlocklist];
  renderBlocklist();
  blockAllShortsCheckbox.checked = result[STORAGE_KEYS.blockAllShorts] === true;
  cleanSidebarCheckbox.checked = result[STORAGE_KEYS.cleanSidebar] === true;
  backgroundPlaybackCheckbox.checked = result[STORAGE_KEYS.backgroundPlayback] === true;
  if (forceDesktopModeCheckbox) forceDesktopModeCheckbox.checked = result[STORAGE_KEYS.forceDesktopMode] === true;
  adBlockToggle.checked = result[STORAGE_KEYS.adBlockEnabled] !== false;
  videoAdsToggle.checked = result[STORAGE_KEYS.blockVideoAds] !== false;
  bannerAdsToggle.checked = result[STORAGE_KEYS.blockBannerAds] !== false;
  updateAdTogglesUI();
}

function updateAdTogglesUI() {
  const enabled = adBlockToggle.checked;
  videoAdsToggle.disabled = !enabled;
  bannerAdsToggle.disabled = !enabled;
}

async function saveBlocklist() {
  await chrome.storage.sync.set({ [STORAGE_KEYS.blockedChannels]: currentBlocklist });
  filteredBlocklist = [...currentBlocklist];
  renderBlocklist();
  await notifyYouTubeTabs(MESSAGE_TYPES.SETTINGS_UPDATED);
}

function renderBlocklist() {
  blocklistUl.innerHTML = '';
  blocklistCount.textContent = filteredBlocklist.length;
  if (filteredBlocklist.length === 0) {
    const empty = document.createElement('li');
    empty.textContent = 'No blocked channels';
    empty.style.justifyContent = 'center';
    empty.style.background = 'transparent';
    blocklistUl.appendChild(empty);
    return;
  }
  filteredBlocklist.forEach((channel, idx) => {
    const li = document.createElement('li');
    const span = document.createElement('span');
    span.textContent = channel;
    const removeBtn = document.createElement('button');
    removeBtn.textContent = 'Remove';
    removeBtn.className = 'remove-btn';
    const originalIndex = currentBlocklist.indexOf(channel);
    removeBtn.addEventListener('click', () => removeChannel(originalIndex));
    li.appendChild(span);
    li.appendChild(removeBtn);
    blocklistUl.appendChild(li);
  });
}

function filterBlocklist() {
  const query = blocklistSearch.value.trim().toLowerCase();
  filteredBlocklist = currentBlocklist.filter(ch => ch.toLowerCase().includes(query));
  renderBlocklist();
}

blocklistSearch.addEventListener('input', filterBlocklist);

async function addChannel() {
  let raw = channelInput.value.trim();
  if (!raw) return;
  let identifier = extractChannelIdentifier(raw);
  if (!identifier) return;
  if (!currentBlocklist.includes(identifier)) {
    currentBlocklist.push(identifier);
    await saveBlocklist();
    channelInput.value = '';
    filterBlocklist();
  } else {
    channelInput.style.borderColor = '#fc8181';
    setTimeout(() => channelInput.style.borderColor = '#e2e8f0', 1000);
  }
}

async function removeChannel(index) {
  currentBlocklist.splice(index, 1);
  await saveBlocklist();
  filterBlocklist();
}

async function clearAllChannels() {
  if (currentBlocklist.length > 0 && confirm('Remove all blocked channels?')) {
    currentBlocklist = [];
    await saveBlocklist();
    filterBlocklist();
  }
}

async function loadStatistics() {
  try {
    const stats = await sendToBackground(MESSAGE_TYPES.GET_STATS);
    if (stats) {
      totalBlocksSpan.textContent = (stats.totalBlocks || 0).toLocaleString();
      const today = stats.dailyStats || { count: 0 };
      todayBlocksSpan.textContent = today.count.toLocaleString();
    }
  } catch (err) { console.error('Failed to load statistics:', err); }
}

async function loadAdStats() {
  const { adsBlockedToday = 0, adsBlockedTotal = 0 } = await chrome.storage.sync.get([
    STORAGE_KEYS.adsBlockedToday, STORAGE_KEYS.adsBlockedTotal
  ]);
  adsBlockedTodaySpan.textContent = adsBlockedToday;
  adsBlockedTotalSpan.textContent = adsBlockedTotal;
}

// Tab switching
function initTabs() {
  const tabs = document.querySelectorAll('.tab-btn');
  const panes = document.querySelectorAll('.tab-pane');
  tabs.forEach(btn => {
    btn.addEventListener('click', () => {
      const tabId = btn.dataset.tab;
      tabs.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      panes.forEach(pane => pane.classList.remove('active'));
      document.getElementById(tabId).classList.add('active');
    });
  });
}

// options/options-blocklist.js
// Manages the blocked channels list (add, remove, clear, search, render)

import { STORAGE_KEYS, MESSAGE_TYPES } from '../shared/constants.js';
import { notifyYouTubeTabs } from '../shared/messaging.js';

// DOM elements (assumed to exist in options.html)
const channelInput = document.getElementById('channelInput');
const addChannelBtn = document.getElementById('addChannelBtn');
const blocklistUl = document.getElementById('blocklist');
const blocklistSearch = document.getElementById('blocklistSearch');
const clearAllChannelsBtn = document.getElementById('clearAllChannelsBtn');
const blocklistCountSpan = document.getElementById('blocklistCount');

let currentBlocklist = [];
let filteredBlocklist = [];

/**
 * Extract a canonical channel identifier from user input
 * Supports: channel name, @handle, or full YouTube URL
 */
function extractChannelIdentifier(input) {
  let trimmed = input.trim();
  if (!trimmed) return null;
  if (trimmed.includes('youtube.com') || trimmed.includes('youtu.be')) {
    try {
      let url = new URL(trimmed);
      let match = url.pathname.match(/\/@([^/?]+)/);
      if (match) return '@' + match[1];
      match = url.pathname.match(/\/c\/([^/?]+)/);
      if (match) return match[1];
      match = url.pathname.match(/\/channel\/([^/?]+)/);
      if (match) return match[1];
    } catch (e) {}
  }
  if (trimmed.startsWith('@')) return trimmed;
  return trimmed;
}

/**
 * Render the blocked channels list in the UI (uses filtered list)
 */
function renderBlocklist() {
  blocklistUl.innerHTML = '';
  blocklistCountSpan.textContent = filteredBlocklist.length;
  if (filteredBlocklist.length === 0) {
    const empty = document.createElement('li');
    empty.textContent = 'No blocked channels';
    empty.style.justifyContent = 'center';
    empty.style.background = 'transparent';
    blocklistUl.appendChild(empty);
    return;
  }
  filteredBlocklist.forEach((channel, idx) => {
    const li = document.createElement('li');
    const span = document.createElement('span');
    span.textContent = channel;
    const removeBtn = document.createElement('button');
    removeBtn.textContent = 'Remove';
    removeBtn.className = 'remove-item';
    // Find original index in currentBlocklist for removal
    const originalIndex = currentBlocklist.indexOf(channel);
    removeBtn.addEventListener('click', () => removeChannel(originalIndex));
    li.appendChild(span);
    li.appendChild(removeBtn);
    blocklistUl.appendChild(li);
  });
}

/**
 * Update filtered list based on search query and re-render
 */
function filterBlocklist() {
  const query = blocklistSearch.value.trim().toLowerCase();
  filteredBlocklist = currentBlocklist.filter(ch => ch.toLowerCase().includes(query));
  renderBlocklist();
}

/**
 * Save the current blocklist to storage and notify YouTube tabs
 */
async function saveBlocklist() {
  await chrome.storage.sync.set({ [STORAGE_KEYS.blockedChannels]: currentBlocklist });
  filteredBlocklist = [...currentBlocklist];
  renderBlocklist();
  await notifyYouTubeTabs(MESSAGE_TYPES.SETTINGS_UPDATED);
}

/**
 * Add a new channel to the blocklist
 */
export async function addChannel() {
  let raw = channelInput.value.trim();
  if (!raw) return;
  let identifier = extractChannelIdentifier(raw);
  if (!identifier) return;
  if (!currentBlocklist.includes(identifier)) {
    currentBlocklist.push(identifier);
    await saveBlocklist();
    channelInput.value = '';
    filterBlocklist(); // re-apply search
  } else {
    channelInput.style.borderColor = '#fc8181';
    setTimeout(() => channelInput.style.borderColor = '#e2e8f0', 1000);
  }
}

/**
 * Remove a channel at a specific index
 */
async function removeChannel(index) {
  currentBlocklist.splice(index, 1);
  await saveBlocklist();
  filterBlocklist();
}

/**
 * Clear the entire blocklist
 */
export async function clearAllChannels() {
  if (currentBlocklist.length > 0 && confirm('Clear all blocked channels?')) {
    currentBlocklist = [];
    await saveBlocklist();
    filterBlocklist();
  }
}

/**
 * Load the blocklist from storage
 */
export async function loadBlocklist() {
  const result = await chrome.storage.sync.get(STORAGE_KEYS.blockedChannels);
  currentBlocklist = result[STORAGE_KEYS.blockedChannels] || [];
  filteredBlocklist = [...currentBlocklist];
  renderBlocklist();
  if (blocklistSearch) {
    blocklistSearch.value = '';
    blocklistSearch.addEventListener('input', filterBlocklist);
  }
}

// Attach event listeners (called from options.js)
addChannelBtn.addEventListener('click', addChannel);
channelInput.addEventListener('keypress', (e) => {
  if (e.key === 'Enter') addChannel();
});
clearAllChannelsBtn.addEventListener('click', clearAllChannels);

let keywordBlockingEnabled = false;

/**
 * Render the keyword list in the UI
 */
function renderKeywordList() {
  keywordList.innerHTML = '';
  if (currentKeywords.length === 0) {
    const empty = document.createElement('li');
    empty.textContent = 'No keywords';
    empty.style.justifyContent = 'center';
    empty.style.background = 'transparent';
    keywordList.appendChild(empty);
    return;
  }
  currentKeywords.forEach((kw, idx) => {
    const li = document.createElement('li');
    li.textContent = kw;
    const removeBtn = document.createElement('button');
    removeBtn.textContent = 'Remove';
    removeBtn.className = 'remove-item';
    removeBtn.addEventListener('click', () => removeKeyword(idx));
    li.appendChild(removeBtn);
    keywordList.appendChild(li);
  });
}

/**
 * Save keywords to storage and notify
 */
async function saveKeywords() {
  await chrome.storage.sync.set({ [STORAGE_KEYS.blockKeywords]: currentKeywords });
  renderKeywordList();
  await notifyYouTubeTabs(MESSAGE_TYPES.SETTINGS_UPDATED);
}

/**
 * Save keyword blocking enabled state
 */
async function saveKeywordBlockingEnabled(enabled) {
  await chrome.storage.sync.set({ [STORAGE_KEYS.enableKeywordBlocking]: enabled });
  await notifyYouTubeTabs(MESSAGE_TYPES.SETTINGS_UPDATED);
}

/**
 * Add a single keyword
 */
async function addKeyword() {
  let kw = keywordInput.value.trim().toLowerCase();
  if (!kw) return;
  if (!currentKeywords.includes(kw)) {
    currentKeywords.push(kw);
    await saveKeywords();
    keywordInput.value = '';
  } else {
    keywordInput.style.borderColor = '#fc8181';
    setTimeout(() => keywordInput.style.borderColor = '#e2e8f0', 1000);
  }
}

/**
 * Add multiple keywords from textarea (comma separated)
 */
async function addBulkKeywords() {
  let text = bulkKeywords.value.trim();
  if (!text) return;
  const words = text.split(',').map(w => w.trim().toLowerCase()).filter(w => w.length > 0);
  let added = false;
  for (const word of words) {
    if (!currentKeywords.includes(word)) {
      currentKeywords.push(word);
      added = true;
    }
  }
  if (added) {
    await saveKeywords();
    bulkKeywords.value = '';
  } else {
    bulkKeywords.style.borderColor = '#fc8181';
    setTimeout(() => bulkKeywords.style.borderColor = '#e2e8f0', 1000);
  }
}

/**
 * Remove keyword by index
 */
async function removeKeyword(index) {
  currentKeywords.splice(index, 1);
  await saveKeywords();
}

/**
 * Clear all keywords
 */
async function clearAllKeywords() {
  if (currentKeywords.length > 0 && confirm('Clear all keywords?')) {
    currentKeywords = [];
    await saveKeywords();
  }
}

/**
 * Toggle keyword blocking on/off
 */
async function toggleKeywordBlocking() {
  keywordBlockingEnabled = enableKeywordBlockingCheckbox.checked;
  await saveKeywordBlockingEnabled(keywordBlockingEnabled);
}

/**
 * Load keywords from storage
 */
export async function loadKeywords() {
  const result = await chrome.storage.sync.get([
    STORAGE_KEYS.blockKeywords,
    STORAGE_KEYS.enableKeywordBlocking
  ]);
  currentKeywords = result[STORAGE_KEYS.blockKeywords] || [];
  keywordBlockingEnabled = result[STORAGE_KEYS.enableKeywordBlocking] === true;
  enableKeywordBlockingCheckbox.checked = keywordBlockingEnabled;
  renderKeywordList();
}

/**
 * Attach event listeners for keyword controls
 */
export function initKeywordListeners() {
  addKeywordBtn.addEventListener('click', addKeyword);
  keywordInput.addEventListener('keypress', (e) => {
    if (e.key === 'Enter') addKeyword();
  });
  clearKeywordsBtn.addEventListener('click', clearAllKeywords);
  addBulkKeywordsBtn.addEventListener('click', addBulkKeywords);
  enableKeywordBlockingCheckbox.addEventListener('change', toggleKeywordBlocking);
}

// options/options-stats.js
// Manages statistics display (total, today, week) and reset functionality

import { MESSAGE_TYPES } from '../shared/constants.js';
import { sendToBackground } from '../shared/messaging.js';

// DOM elements (assumed to exist in options.html)
const totalBlocksSpan = document.getElementById('totalBlocksOptions');
const todayBlocksSpan = document.getElementById('todayBlocksOptions');
const weekBlocksSpan = document.getElementById('weekBlocksOptions');
const resetStatsBtn = document.getElementById('resetStatsBtn');

/**
 * Load statistics from background and update the UI
 */
export async function loadStatistics() {
  try {
    const stats = await sendToBackground(MESSAGE_TYPES.GET_STATS);
    if (stats) {
      totalBlocksSpan.textContent = (stats.totalBlocks || 0).toLocaleString();
      const today = stats.dailyStats || { count: 0 };
      todayBlocksSpan.textContent = today.count.toLocaleString();
      const weekly = stats.weeklyStats || [];
      const weekTotal = weekly.reduce((sum, day) => sum + day.count, 0);
      weekBlocksSpan.textContent = weekTotal.toLocaleString();
    }
  } catch (err) {
    console.error('Failed to load statistics:', err);
  }
}

/**
 * Reset all statistics (total, daily, weekly) after confirmation
 */
export async function resetStatistics() {
  if (confirm('Reset all block statistics? This cannot be undone.')) {
    const today = new Date().toISOString().split('T')[0];
    await chrome.storage.sync.set({
      totalBlocks: 0,
      dailyStats: { date: today, count: 0 },
      weeklyStats: []
    });
    // Clear the badge number
    chrome.action.setBadgeText({ text: '' });
    // Reload the UI
    await loadStatistics();
  }
}

/**
 * Attach event listeners for statistics
 */
export function initStatsListeners() {
  resetStatsBtn.addEventListener('click', resetStatistics);
}

// options/options-encryption.js
import { STORAGE_KEYS } from '../shared/constants.js';
import { encryptData, decryptData } from '../shared/encryption.js';
import { notifyYouTubeTabs } from '../shared/messaging.js';

const encryptionToggle = document.getElementById('encryptionToggle');
const encryptionPassphraseInput = document.getElementById('encryptionPassphrase');
const applyEncryptionBtn = document.getElementById('applyEncryptionBtn');

// Helper: generate random salt (16 bytes)
function generateSalt() {
  const array = new Uint8Array(16);
  crypto.getRandomValues(array);
  return Array.from(array).map(b => b.toString(16).padStart(2, '0')).join('');
}

// Collect all sensitive data (blocklist, whitelist, keywords, etc.)
async function collectSensitiveData() {
  const keys = [
    STORAGE_KEYS.blockedChannels,
    STORAGE_KEYS.blockKeywords,
    STORAGE_KEYS.whitelistChannels,
    STORAGE_KEYS.whitelistKeywords,
    STORAGE_KEYS.whitelistSearchKeywords,
    STORAGE_KEYS.whitelistSearchSelected
  ];
  const data = await chrome.storage.sync.get(keys);
  return data;
}

// Encrypt and store encrypted data, then clear plain text
async function enableEncryption(passphrase) {
  const salt = generateSalt();
  const data = await collectSensitiveData();
  const encrypted = await encryptData(data, passphrase, salt);
  await chrome.storage.sync.set({
    [STORAGE_KEYS.encryptionEnabled]: true,
    [STORAGE_KEYS.encryptionSalt]: salt,
    [STORAGE_KEYS.encryptedData]: encrypted
  });
  // Clear plain text sensitive data
  const clearKeys = [
    STORAGE_KEYS.blockedChannels,
    STORAGE_KEYS.blockKeywords,
    STORAGE_KEYS.whitelistChannels,
    STORAGE_KEYS.whitelistKeywords,
    STORAGE_KEYS.whitelistSearchKeywords,
    STORAGE_KEYS.whitelistSearchSelected
  ];
  for (const key of clearKeys) {
    await chrome.storage.sync.set({ [key]: [] }); // or default empty
  }
  // Store passphrase in chrome.storage.local (not synced)
  await chrome.storage.local.set({ encryptionPassphrase: passphrase });
  console.log('Encryption enabled');
}

// Decrypt data and restore to sync storage
async function disableEncryption(passphrase) {
  const { encryptionSalt, encryptedData } = await chrome.storage.sync.get([
    STORAGE_KEYS.encryptionSalt,
    STORAGE_KEYS.encryptedData
  ]);
  if (!encryptionSalt || !encryptedData) {
    console.error('No encrypted data found');
    return false;
  }
  try {
    const decrypted = await decryptData(encryptedData, passphrase, encryptionSalt);
    // Restore decrypted data to sync storage
    for (const [key, value] of Object.entries(decrypted)) {
      await chrome.storage.sync.set({ [key]: value });
    }
    // Remove encryption flags
    await chrome.storage.sync.remove([
      STORAGE_KEYS.encryptionEnabled,
      STORAGE_KEYS.encryptionSalt,
      STORAGE_KEYS.encryptedData
    ]);
    await chrome.storage.local.remove('encryptionPassphrase');
    console.log('Encryption disabled');
    return true;
  } catch (err) {
    console.error('Decryption failed: wrong passphrase?', err);
    return false;
  }
}

async function applyEncryption() {
  const passphrase = encryptionPassphraseInput.value.trim();
  if (!passphrase) {
    alert('Please enter a passphrase');
    return;
  }
  const isEnabled = encryptionToggle.checked;
  if (isEnabled) {
    await enableEncryption(passphrase);
    alert('Encryption enabled. Sensitive data is now encrypted.');
  } else {
    const success = await disableEncryption(passphrase);
    if (success) {
      alert('Encryption disabled. Data restored.');
    } else {
      alert('Failed to decrypt. Check your passphrase.');
      encryptionToggle.checked = true; // revert toggle
    }
  }
  // Reload options page to reflect changes
  window.location.reload();
}

export async function loadEncryptionSettings() {
  const { encryptionEnabled = false } = await chrome.storage.sync.get(STORAGE_KEYS.encryptionEnabled);
  encryptionToggle.checked = encryptionEnabled;
}

export function initEncryptionListeners() {
  applyEncryptionBtn.addEventListener('click', applyEncryption);
}

import { MESSAGE_TYPES } from './constants.js';

export async function sendToBackground(type, payload = {}) {
  return new Promise((resolve, reject) => {
    chrome.runtime.sendMessage({ type, ...payload }, (response) => {
      if (chrome.runtime.lastError) {
        reject(new Error(chrome.runtime.lastError.message));
      } else {
        resolve(response);
      }
    });
  });
}

export async function notifyYouTubeTabs(type, payload = {}) {
  const tabs = await chrome.tabs.query({ url: '*://www.youtube.com/*' });
  for (const tab of tabs) {
    try {
      await chrome.tabs.sendMessage(tab.id, { type, ...payload });
    } catch (err) {}
  }
}

export function addMessageListener(callback) {
  chrome.runtime.onMessage.addListener(callback);
}

export function createSettingsUpdatedMessage() {
  return { type: MESSAGE_TYPES.SETTINGS_UPDATED };
}

// shared/encryption.js
export async function deriveKey(passphrase, salt) {
  const enc = new TextEncoder();
  const keyMaterial = await crypto.subtle.importKey(
    'raw', enc.encode(passphrase), 'PBKDF2', false, ['deriveBits', 'deriveKey']
  );
  return crypto.subtle.deriveKey(
    { name: 'PBKDF2', salt: enc.encode(salt), iterations: 100000, hash: 'SHA-256' },
    keyMaterial,
    { name: 'AES-GCM', length: 256 },
    false,
    ['encrypt', 'decrypt']
  );
}

export async function encryptData(data, passphrase, salt) {
  const key = await deriveKey(passphrase, salt);
  const enc = new TextEncoder();
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const encrypted = await crypto.subtle.encrypt(
    { name: 'AES-GCM', iv },
    key,
    enc.encode(JSON.stringify(data))
  );
  return { iv: Array.from(iv), ciphertext: Array.from(new Uint8Array(encrypted)) };
}

export async function decryptData(encryptedObj, passphrase, salt) {
  const key = await deriveKey(passphrase, salt);
  const iv = new Uint8Array(encryptedObj.iv);
  const ciphertext = new Uint8Array(encryptedObj.ciphertext);
  const decrypted = await crypto.subtle.decrypt(
    { name: 'AES-GCM', iv },
    key,
    ciphertext
  );
  return JSON.parse(new TextDecoder().decode(decrypted));
}
